   512→  isRtl: boolean;
   513→  lang: "ar" | "en";
   514→  theme: "light" | "dark";
   515→  setLang: (l: "ar" | "en") => void;
   516→  setTheme: (t: "light" | "dark") => void;
   517→  onClose?: () => void;
   518→}) {
   519→  const roleSpecificItems = navItems.filter((n) => n.roleSpecific);
   520→  const commonItems = navItems.filter((n) => !n.roleSpecific);
   521→
   522→  return (
   523→    <>
   524→      {/* Logo header */}
   525→      <div className="h-20 flex items-center justify-between px-4 border-b border-sidebar-border/80 flex-shrink-0">
   526→        {sidebarOpen ? (
   527→          <BrandLogo size="sm" />
   528→        ) : (
   529→          <div className="h-9 w-9 rounded-xl bg-gradient-to-br from-neutral-950 via-zinc-800 to-zinc-600 flex items-center justify-center mx-auto">
   530→            <svg viewBox="0 0 24 24" className="h-5 w-5 text-white" fill="none" stroke="currentColor" strokeWidth="2.5">
   531→              <path d="M3 21h18M5 21V8l7-5 7 5v13M9 21v-6h6v6" strokeLinecap="round" strokeLinejoin="round" />
   532→            </svg>
   533→          </div>
   534→        )}
   535→        {onClose && (
   536→          <Button variant="ghost" size="icon" onClick={onClose} className="lg:hidden">
   537→            <X className="h-5 w-5" />
   538→          </Button>
   539→        )}
   540→      </div>
   541→
   542→      {/* Nav */}
   543→      <nav className="flex-1 overflow-y-auto scrollbar-thin px-3 py-5 space-y-1.5">
   544→        {commonItems.map((item) => (
   545→          <NavButton
   546→            key={item.key}
   547→            item={item}
   548→            active={currentView === item.view}
   549→            onClick={() => setView(item.view as any)}
   550→            sidebarOpen={sidebarOpen}
   551→            lang={lang}
   552→          />
   553→        ))}
   554→
   555→        {roleSpecificItems.length > 0 && (
   556→          <>
   557→            <div className="pt-4 pb-1">
   558→              {sidebarOpen && (
   559→                <p className="text-[10px] uppercase tracking-wider text-muted-foreground/70 font-semibold px-3 mb-1">
   560→                  {role === "bundle_creator" ? (lang === "ar" ? "الباقات" : "Bundles") : (lang === "ar" ? "الفندق" : "Property")}
   561→                </p>
   562→              )}
   563→              {roleSpecificItems.map((item) => (
   564→                <NavButton
   565→                  key={item.key}
   566→                  item={item}
   567→                  active={currentView === item.view}
   568→                  onClick={() => setView(item.view as any)}
   569→                  sidebarOpen={sidebarOpen}
   570→                  lang={lang}
   571→                />
   572→              ))}
   573→            </div>
   574→          </>
   575→        )}
   576→      </nav>
   577→
   578→      {/* Footer */}
   579→      <div className="border-t border-sidebar-border p-3 space-y-1.5 flex-shrink-0">
   580→        <button
   581→          onClick={() => setLang(lang === "ar" ? "en" : "ar")}
   582→          className="w-full flex items-center gap-3 px-3 py-2.5 rounded-xl text-sm hover:bg-sidebar-accent transition-colors text-sidebar-foreground"
   583→        >
   584→          <Globe className="h-4 w-4 flex-shrink-0" />
   585→          {sidebarOpen && <span>{t("language_toggle", lang)}</span>}
   586→        </button>
   587→        <button
   588→          onClick={() => setTheme(theme === "dark" ? "light" : "dark")}
   589→          className="w-full flex items-center gap-3 px-3 py-2.5 rounded-xl text-sm hover:bg-sidebar-accent transition-colors text-sidebar-foreground"
   590→        >
   591→          {theme === "dark" ? <Sun className="h-4 w-4 flex-shrink-0" /> : <Moon className="h-4 w-4 flex-shrink-0" />}
   592→          {sidebarOpen && (
   593→            <span>
   594→              {theme === "dark"
   595→                ? (lang === "ar" ? "الوضع الفاتح" : "Light Mode")
   596→                : (lang === "ar" ? "الوضع الداكن" : "Dark Mode")}
   597→            </span>
   598→          )}
   599→        </button>
   600→        <button
   601→          onClick={onLogout}
   602→          className="w-full flex items-center gap-3 px-3 py-2.5 rounded-xl text-sm hover:bg-destructive/10 hover:text-destructive transition-colors text-sidebar-foreground"
   603→        >
   604→          <LogOut className="h-4 w-4 flex-shrink-0" />
   605→          {sidebarOpen && <span>{t("logout", lang)}</span>}
   606→        </button>
   607→      </div>
   608→    </>
   609→  );
   610→}
   611→
   612→function NavButton({
   613→  item,
   614→  active,
   615→  onClick,
   616→  sidebarOpen,
   617→  lang,
   618→}: {
   619→  item: NavItem;
   620→  active: boolean;
   621→  onClick: () => void;
   622→  sidebarOpen: boolean;
   623→  lang: "ar" | "en";
   624→}) {
   625→  const Icon = item.icon;
   626→  return (
   627→    <button
   628→      onClick={onClick}
   629→      title={!sidebarOpen ? t(item.labelKey, lang) : undefined}
   630→      className={cn(
   631→        "w-full flex items-center gap-3 px-3 py-2.5 rounded-xl text-sm transition-all relative group",
   632→        active
   633→          ? "bg-primary text-primary-foreground shadow-[0_12px_26px_-18px_rgba(15,23,42,0.6)]"
   634→          : "text-sidebar-foreground hover:bg-sidebar-accent",
   635→        !sidebarOpen && "justify-center"
   636→      )}
   637→    >
   638→      <Icon className={cn("h-4.5 w-4.5 flex-shrink-0", !active && "text-muted-foreground group-hover:text-foreground")} style={{ width: "1.125rem", height: "1.125rem" }} />
   639→      {sidebarOpen && <span className="font-medium">{t(item.labelKey, lang)}</span>}
   640→      {!sidebarOpen && active && (
   641→        <span className="absolute -end-3 top-1/2 -translate-y-1/2 h-6 w-1 rounded-s-full bg-primary" />
   642→      )}
   643→    </button>
   644→  );
   645→}
